% --- Convenience commands ---
\newcommand{\trendUp}{\textcolor{errred}{$\blacktriangle$}}
\newcommand{\trendDown}{\textcolor{okgreen}{$\blacktriangledown$}}
\newcommand{\trendStable}{\textcolor{gray}{$\bullet$}}

\newcommand{\risklabel}[1]{%
  \ifdim #1pt > 40pt
    \textcolor{errred}{\textbf{#1\%}}%
  \else\ifdim #1pt > 25pt
    \textcolor{warnblue}{#1\%}%
  \else
    \textcolor{okgreen}{#1\%}%
  \fi\fi
}

% Statistical significance marker
\newcommand{\sigmark}[1]{%
  \ifdim #1pt < 0.001pt
    \textcolor{errred}{\textbf{***}}%
  \else\ifdim #1pt < 0.01pt
    \textcolor{errred}{\textbf{**}}%
  \else\ifdim #1pt < 0.05pt
    \textcolor{warnblue}{\textbf{*}}%
  \else
    \textcolor{gray}{ns}%
  \fi\fi\fi
}

% Highlight box for key metrics
\newcommand{\metricbox}[2]{%
  \begin{beamercolorbox}[rounded=true,shadow=true,wd=0.45\textwidth,sep=5pt]{block body}
    \centering
    {\small\color{bmadark}#1}\\[3pt]
    {\Large\bfseries\color{bmagreen}#2}
  \end{beamercolorbox}%
}
